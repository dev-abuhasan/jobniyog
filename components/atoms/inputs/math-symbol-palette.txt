"use client";

import React from 'react';

interface MathSymbolPaletteProps {
    onInsert: (symbol: string) => void;
}

export const MathSymbolPalette = ({ onInsert }: MathSymbolPaletteProps) => {
    const symbolCategories = {
        Greek: ['α', 'β', 'γ', 'δ', 'ε', 'ζ', 'η', 'θ', 'ι', 'κ', 'λ', 'μ', 'ν', 'ξ', 'ο', 'π', 'ρ', 'σ', 'τ', 'υ', 'φ', 'χ', 'ψ', 'ω', 'Γ', 'Δ', 'Θ', 'Λ', 'Ξ', 'Π', 'Σ', 'Φ', 'Ψ', 'Ω'],
        Operators: ['+', '−', '×', '÷', '±', '∓', '∗', '∘', '⋅', '⊕', '⊗', '⊖', '∧', '∨', '¬', '∀', '∃', '∄'],
        Relations: ['=', '≠', '≈', '≡', '≅', '≤', '≥', '⊂', '⊃', '⊆', '⊇', '∈', '∉', '∋', '∌', '∝', '∞', '∅', 'ℕ', 'ℤ', 'ℚ', 'ℝ', 'ℂ'],
        Arrows: ['→', '←', '↔', '⇒', '⇐', '⇔', '↦', '↻', '↺', '⟶', '⟵', '⟷', '⟹', '⟸', '⟺'],
        'Set Theory': ['∪', '∩', '⊓', '⊔', '∖', '△', '▽', '◁', '▷', '⋃', '⋂', '⋀', '⋁'],
        Calculus: ['∫', '∬', '∭', '∮', '∂', '∇', '∆', '∑', '∏', 'lim', 'sup', 'inf'],
        Misc: ['√', '∛', '∜', '∞', 'ℵ', 'ℏ', 'ℓ', '℘', 'ℑ', 'ℜ']
    };

    return (
        <div className="bg-white border border-gray-200 rounded-lg shadow-lg p-3 max-w-3xl max-h-64 overflow-y-auto">
            {Object.entries(symbolCategories).map(([category, symbols]) => (
                <div key={category} className="mb-3">
                    <h5 className="text-xs font-semibold text-gray-500 uppercase tracking-wider mb-1">{category}</h5>
                    <div className="flex flex-wrap gap-1">
                        {symbols.map((symbol) => (
                            <button
                                key={symbol}
                                onClick={() => onInsert(`$${symbol}$`)}
                                className="w-8 h-8 text-sm bg-gray-50 hover:bg-blue-50 border border-gray-200 rounded transition-colors font-mono"
                                title={symbol}
                                type="button"
                            >
                                {symbol}
                            </button>
                        ))}
                    </div>
                </div>
            ))}
        </div>
    );
};
